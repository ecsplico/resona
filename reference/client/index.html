    resona-client¶

    resona-client is the Python client library for talking to a resona-api server. It handles authentication, job submission, polling, and full CRUD for postprocessing profiles. Engine configuration — which servers to connect to, auto-start rules, and cloud provider registrations — lives in ~/.resona/config.json and is managed through EngineConfig and EngineEntry.

    Installation¶

    resona-client is a workspace package. Add it from the monorepo:

    uv add resona-client --package my-project

    Quick example¶

    from resona_client.client import ResonaClient

# Connect using RESONA_API_URL / RESONA_API_KEY env vars (RESONA_API_KEY = org-bound token)
client = ResonaClient()

# Or use the resona-api entry in ~/.resona/config.json
client = ResonaClient.from_config()

# Upload + create job (private by default); returns JobRef(job_id, status)
ref = client.submit_job("recording.wav", kind="audio")
job = client.wait_for_job(ref.job_id)
if job.status == "completed":
    result = client.get_result(ref.job_id)
    print(result.raw_text)   # raw transcript
    print(result.markdown)   # after the postprocess profile

# Submit with a postprocess profile stored on the hub
ref = client.submit_job("recording.wav", kind="audio", profile="medical-de")

    Authentication¶

    The client sends Authorization: Bearer <token> with every request when api_key is provided or RESONA_API_KEY is set. Tokens are org-bound and issued on the hub with resona-admin credential issue --organization <id> --role client. The hub derives the organization from the token; the client never sends a tenant identifier.

    client = ResonaClient(base_url="https://resona.example.com", api_key="<token>")

    Job lifecycle¶

    Jobs progress through pending → processing → completed | failed | cancelled. wait_for_job() polls until a terminal state and raises JobTimeout when its timeout passes; the job keeps running on the hub and the client never falls back to local processing. Each submit_job() call sends one Idempotency-Key and reuses it on retries, so a lost response cannot create a duplicate job.

    ref = client.submit_job("dictation.mp3", kind="audio")
job = client.wait_for_job(ref.job_id, timeout=600)

if job.status == "completed":
    print(client.get_result(ref.job_id).markdown)
else:
    print("not completed:", job.status, job.error)

client.cancel_job(ref.job_id)   # fence the lease
client.delete_job(ref.job_id)   # delete input and result now

    get_result() raises JobNotReady (409) before completion and ResultExpired (410) after the 24-hour result retention.

    For content already in memory (the MCP server's upload_content), upload and submit in two steps; the upload is consumed by exactly one job:

    up = client.upload_bytes("memo.wav", data)            # POST /v1/uploads → Upload(upload_id, sha256, size)
ref = client.submit_upload(up.upload_id, kind="audio", execution_profile="cpu-private")

    v1 Audio routes¶

    The v1 routes call engines synchronously through the gateway and return results immediately — no job queue involved.

    # Transcribe synchronously
result = client.create_transcription("file.wav", engine="faster-whisper", language="de")
print(result["text"])

# Transcribe with verbose output (segments + duration)
result = client.create_transcription("file.wav", response_format="verbose_json")
print(result["segments"])

# Text-to-speech
audio_bytes = client.create_speech("Hallo Welt", voice="alloy", engine="openai")
with open("out.mp3", "wb") as f:
    f.write(audio_bytes)

# List available engines
engines = client.list_engines()

    Profile management¶

    # List all profiles stored on the server
profiles = client.list_profiles()
for p in profiles:
    print(p["name"], p["description"])

# Get the full JSON of a profile
profile = client.get_profile("medical-de")

# Upload a local profile to the server
import json
from pathlib import Path
profile_data = json.loads(Path("my-profile.json").read_text())
client.push_profile(profile_data)

# Download a profile from the server
profile = client.pull_profile("medical-de")
Path("medical-de.json").write_text(json.dumps(profile, ensure_ascii=False, indent=2))

# Delete a profile from the server
client.delete_profile("old-profile")

    Engine configuration¶

    EngineConfig and EngineEntry describe which servers the client knows about. Manage them with resona engines add/remove/list from the CLI; or load and modify them programmatically.

    from resona_client.config import EngineConfig, EngineEntry, resolve_engine

cfg = EngineConfig.load()
print(cfg.default_engine)       # "faster-whisper"
print([e.name for e in cfg.engines])

# Check which entry would be used right now
entry = resolve_engine()
if entry:
    print(entry.api_url)

    Context manager¶

    ResonaClient implements the context-manager protocol and closes the underlying httpx connection pool on exit:

    with ResonaClient() as client:
    ref = client.submit_job("audio.wav", kind="audio")
    job = client.wait_for_job(ref.job_id)

    @classmethod
def from_config(cls, auto_start: bool = True, timeout: float = 3600.0) -> "ResonaClient":
    """
    Create a client by resolving the engine to use:

    1. RESONA_API_URL env var (if set) — used directly, no config lookup.
    2. WS_API_URL env var (if set) — used directly, no config lookup (fallback).
    3. First reachable engine in ~/.resona/config.json.
    4. If none reachable and an engine has compose_dir set, start it via
       docker compose up -d and wait for it to become healthy.

    Raises RuntimeError if no engine could be resolved.
    """
    env_url = os.getenv("RESONA_API_URL") or os.getenv("WS_API_URL")
    if env_url:
        return cls(base_url=env_url, timeout=timeout)

    from .config import resolve_engine
    entry = resolve_engine(auto_start=auto_start)
    if entry:
        return cls(base_url=entry.api_url, api_key=entry.api_key, timeout=timeout)

    raise RuntimeError(
        "No reachable resona engine found.\n"
        "Add one with:  resona engines add <name> <url>"
    )

    def submit_upload(
    self,
    upload_id: str,
    *,
    kind: JobKind,
    execution_profile: Optional[str] = None,
    language: Optional[str] = None,
    engine: Optional[str] = None,
    model: Optional[str] = None,
    profile: Optional[str] = None,
    options: Union[AudioOptions, DocumentOptions, Mapping[str, Any], None] = None,
    private: bool = True,
    idempotency_key: Optional[str] = None,
    provenance: Optional[Mapping[str, Any]] = None,
    purpose: JobPurpose = "production",
    release_id: Optional[str] = None,
) -> JobRef:
    """Create a job for an existing upload. POST /v1/jobs

    The Idempotency-Key (generated unless given) is reused for every retry of
    the POST, so a lost response cannot create a duplicate job.
    ``purpose='evaluation'`` needs an org-admin credential and the ``release_id``
    of a release staged on the hub (``resona-admin release stage``).
    """
    checked = _checked_options(kind, options)
    key = idempotency_key or str(uuid.uuid4())
    body: dict[str, Any] = {
        "kind": kind,
        "upload_id": upload_id,
        "private": private,
    }
    optional: dict[str, Any] = {
        "execution_profile": execution_profile,
        "language": language,
        "engine": engine,
        "model": model,
        "profile": profile,
        "options": checked,
        "provenance": dict(provenance) if provenance is not None else None,
        "purpose": purpose if purpose != "production" else None,
        "release_id": release_id,
    }
    body.update({k: v for k, v in optional.items() if v is not None})

    resp = self._post_job(body, key)
    data = resp.json()
    return JobRef(job_id=int(data["job_id"]), status=str(data["status"]))

    def list_jobs(self, *, cursor: Optional[str] = None, limit: int = 50) -> JobPage:
    """List the organization's jobs, newest first. GET /v1/jobs"""
    if not 1 <= limit <= 100:
        raise ValueError("limit must be between 1 and 100")
    params: dict[str, Any] = {"limit": limit}

    body = self._send("GET", "/v1/jobs", params=params).json()
    return JobPage(

        next_cursor=body.get("next_cursor"),
    )

    def wait_for_job(
    self,
    job_id: int,
    *,
    timeout: Optional[float] = None,
    poll: float = 1.0,
) -> JobView:
    """Poll until the job is completed, failed or cancelled.

    Raises JobTimeout when the client deadline passes. The job keeps running on
    the hub; the caller decides what to do. Never falls back to local processing.
    """
    deadline = None if timeout is None else time.monotonic() + timeout
    while True:
        job = self.get_job(job_id)
        if job.terminal:
            return job
        if deadline is None:
            time.sleep(poll)
            continue
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            raise JobTimeout(job_id)
        time.sleep(min(poll, remaining))

    def get_baseline(self, *, execution_profile: str, platform: str) -> Optional[dict]:
    """Current baseline for the profile/platform, or None (hub 404 ``baseline_required``).

    GET /v1/baselines (org_admin)
    """
    resp = self._client.get(
        f"{self.base_url}/v1/baselines",
        params={"execution_profile": execution_profile, "platform": platform},
    )
    if resp.status_code == 404 and _parse_error_body(resp)[0] == "baseline_required":
        return None
    if resp.is_error:
        _raise_for(resp)
    return resp.json()

    def create_transcription(
    self,
    audio_path: "Path | str",
    *,
    model: str = "whisper-1",
    language: Optional[str] = None,
    response_format: str = "json",
    engine: Optional[str] = None,
    execution_profile: Optional[str] = None,
    private: bool = True,
    profile: Optional[str] = None,
) -> dict:
    """Transcribe audio through the OpenAI-compatible adapter. POST /v1/audio/transcriptions

    The hub queues a normal audio job and waits a bounded time for a worker.
    ``model="whisper-1"`` means the execution profile's default; any other value
    is an override that must be an approved candidate.

    Returns:
        Dict with key ``text`` (``json`` and ``text`` formats).
        With ``response_format="verbose_json"``: ``text``, ``language``, ``duration``, ``segments``.

    Raises SyncWaitTimeout (504) when no worker finished in time; the job keeps
    running and ``job_id`` identifies it for ``wait_for_job``/``get_result``.
    """
    audio_path = Path(audio_path)
    data: dict = {
        "model": model,
        "response_format": response_format,
        "private": str(private).lower(),
    }
    optional = {
        "language": language,
        "engine": engine,
        "execution_profile": execution_profile,
        "profile": profile,
    }
    data.update({k: v for k, v in optional.items() if v})
    content_type = mimetypes.guess_type(audio_path.name)[0] or "application/octet-stream"
    with open(audio_path, "rb") as f:
        resp = self._client.post(
            f"{self.base_url}/v1/audio/transcriptions",
            files={"file": (audio_path.name, f, content_type)},
            data=data,
        )
    if resp.status_code == 504:
        code, message = _parse_error_body(resp)
        job_id = _json_field(resp, "job_id")
        if code == "timeout" and isinstance(job_id, int):
            raise SyncWaitTimeout(resp.status_code, code, message, job_id=job_id)
    if resp.is_error:
        _raise_for(resp)
    if response_format == "text":
        return {"text": resp.text}
    return resp.json()

    def resolve_engine(
    auto_start: bool = True,
    connect_timeout: float = 3.0,
    name: Optional[str] = None,
) -> Optional[EngineEntry]:
    """Return a reachable hub entry from ~/.resona/config.json.

    Args:
        auto_start: If True, try to start unreachable entries.
        connect_timeout: Per-entry reachability probe timeout.
        name: If given, resolve only the entry with this exact name.

    Entries are usable when /health responds; if not and ``auto_start`` is
    set, an SSH tunnel or docker compose project is started. A configured
    ``compose_dir`` that does not exist is logged and skipped instead of
    raising FileNotFoundError.

    Returns None if no usable engine is found.
    """
    cfg = EngineConfig.load()
    engines = cfg.engines
    if name is not None:
        engines = [e for e in engines if e.name == name]
    if not engines:
        return None

    # 1. Immediately reachable entries.
    for entry in engines:
        if is_reachable(entry, timeout=connect_timeout):
            return entry

    if not auto_start:
        return None

    # 2. Auto-start entries.
    for entry in engines:
        if entry.ssh_host:
            if entry.name not in _active_tunnels:
                proc = _start_ssh_tunnel(entry)
                _active_tunnels[entry.name] = proc
            host = _local_host(entry.api_url)
            port = _local_port(entry.api_url)
            log.info(f"Waiting for SSH tunnel to '{entry.name}' (up to 30s)...")
            if _wait_for_port(host, port, timeout=30.0) and is_reachable(entry):
                return entry
        elif entry.compose_dir:
            if not Path(entry.compose_dir).is_dir():
                log.warning(
                    f"Engine '{entry.name}': compose_dir '{entry.compose_dir}' "
                    f"does not exist — skipping auto-start."
                )
                continue
            _start_compose(entry.compose_dir)
            log.info(f"Waiting for '{entry.name}' to become available (up to 120s)...")
            if _wait_for_engine(entry):
                return entry

    return None

    def is_reachable(entry: EngineEntry, timeout: float = 3.0) -> bool:
    """Return True if the engine's /health endpoint responds 200."""
    try:
